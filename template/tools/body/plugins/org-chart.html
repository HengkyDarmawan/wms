const data = [
  { id: '1', parentId: '',  name: 'Aigars Silkalns', role: 'Chief Executive Officer', dept: 'Executive' },
  { id: '2', parentId: '1', name: 'Sarah Chen',      role: 'Head of Product',         dept: 'Product'   },
  { id: '3', parentId: '1', name: 'Marcus Webb',     role: 'VP Engineering',          dept: 'Engineering' }
];

new d3.OrgChart()
  .container('#orgChart')
  .data(data)
  .nodeWidth(() => 250)
  .nodeHeight(() => 120)
  .nodeContent(d => `<div class="oc-card">${d.data.name}</div>`)
  .onNodeClick(id => console.log('clicked', id))
  .render();
